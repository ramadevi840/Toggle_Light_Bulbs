class Solution {
    public List<Integer> toggleLightBulbs(List<Integer> bulbs) {
            HashSet<Integer> set=  new HashSet<>();
            for(int i : bulbs){
                if(!set.add(i)){
                    set.remove(i);
                }
            }
        List<Integer> list= new ArrayList<>(set);
        Collections.sort(list);
        return list;   
    }
}
